# ─── Tekscan Pressure Mapping SDK · Python Example ──────────────────────────
# SDK    : Tekscan Pressure Mapping SDK (COM/ActiveX wrapper via comtypes)
# Docs   : https://www.tekscan.com/products-solutions/software/pressure-mapping-software-development-kit-sdk
# System : Animal Strideway — 48×64 sensel array, 100 Hz

import comtypes.client as cc
import numpy as np

# ── 1. Load SDK COM object ────────────────────────────────────────────
sdk = cc.CreateObject("TekscanPressureMapping.API")
sdk.Initialize()

# ── 2. Open sensor ───────────────────────────────────────────────────
sensor = sdk.OpenSensor(sensor_id=0)
sensor.SetFrameRate(100)      # Hz
sensor.SetCalibration(0.42)    # N per sensel per raw unit

rows = sensor.Rows     # 48
cols = sensor.Columns  # 64
print(f"Sensor: {rows}×{cols} sensels @ 100 Hz")

# ── 3. Start acquisition ─────────────────────────────────────────────
sensor.StartAcquisition()

# ── 4. Read frames ────────────────────────────────────────────────────
for frame_idx in range(1000):
    frame = sensor.GetFrame()
    raw   = np.array(frame.Data, dtype=np.float32).reshape((rows, cols))
    force = raw * 0.42         # N per sensel
    area  = np.sum(raw > 0)   # active sensels

    metrics = {
        "frame":        frame_idx,
        "timestamp_ms": frame.TimestampMs,
        "peak_force_N": float(force.max()),
        "total_force_N":float(force.sum()),
        "contact_cm2":  float(area * 0.0154),  # cm² per sensel
        "cop_x":        float(np.average(np.arange(cols), weights=force.sum(axis=0))),
        "cop_y":        float(np.average(np.arange(rows), weights=force.sum(axis=1))),
    }
    print(metrics)

# ── 5. Stop & export ─────────────────────────────────────────────────
sensor.StopAcquisition()
sensor.ExportXML("gait_session.xml")   # standard FScan XML format
sdk.Close()
