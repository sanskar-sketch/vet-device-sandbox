# ─── Clarius Cast API · Python Example ─────────────────────────────────────
# SDK    : pyclariuscast  (from clariusdev/cast releases)
# Docs   : https://github.com/clariusdev/cast
# Prereq : Clarius App must be running and connected to the C7 Vet HD3 probe
# Network: PC and probe on same Wi-Fi (SSID: DIRECT-xxxxxxxx)

import pyclariuscast as clarius
import numpy as np

PROBE_IP   = "192.168.1.1"
CAST_PORT  = 5828
IMG_W, IMG_H = 640, 480

# ── Callbacks ──────────────────────────────────────────────────────────
def new_image_fn(image, width, height, bpp, sz, imu):
    """Called for every new B-mode or Doppler frame."""
    frame = np.frombuffer(image, dtype=np.uint8).reshape((height, width, bpp//8))
    print(f"Frame: {width}×{height}  {bpp}bpp  {sz} bytes  ts={imu.timestamp}")

def new_raw_image_fn(image, lines, samples, bpp, lateral, axial, imu):
    """Called for raw (polar-coordinate) IQ/RF frames when frozen."""
    print(f"Raw: {lines} lines × {samples} samples, lateral={lateral:.2f} mm, axial={axial:.2f} mm")

def freeze_fn(frozen):
    print(f"Imaging {'FROZEN' if frozen else 'LIVE'}")

def button_fn(button, clicks):
    print(f"Probe button {button} pressed {clicks} times")

# ── Initialize Cast API ────────────────────────────────────────────────
ret = clarius.init(
    width      = IMG_W,
    height     = IMG_H,
    new_image  = new_image_fn,
    new_raw    = new_raw_image_fn,
    freeze     = freeze_fn,
    button     = button_fn,
)
if ret != 0:
    raise RuntimeError(f"Cast API init failed: {ret}")

# ── Connect to probe ───────────────────────────────────────────────────
ret = clarius.connect(PROBE_IP, CAST_PORT)
if ret != 0:
    raise RuntimeError(f"Connection failed: {ret}")
print("Connected — streaming B-mode images")

# ── Adjust imaging parameters ──────────────────────────────────────────
clarius.user_function("depth", 8)    # depth = 8 cm
clarius.user_function("gain",  50)   # gain  = 50%

# ── Run until Enter pressed ────────────────────────────────────────────
input("Press Enter to freeze and capture raw data...\n")
clarius.user_function("freeze", 1)

# ── Request raw IQ data (probe must be frozen) ─────────────────────────
clarius.request_raw_data()   # triggers new_raw_image_fn callback

# ── Disconnect ────────────────────────────────────────────────────────
clarius.disconnect()
clarius.destroy()
