// ─── Orbbec Femto Mega · SDK Request Example (Python) ─────────────────────
// SDK    : pyorbbecsdk v2   pip install pyorbbecsdk
// Docs   : https://github.com/orbbec/pyorbbecsdk

import pyorbbecsdk as obs
from pyorbbecsdk import Context, Pipeline, Config, OBSensorType, OBFormat
import numpy as np

# ── 1. Discover device ─────────────────────────────────────────────────
ctx    = obs.Context()

# USB mode
device = ctx.query_devices().get_device_by_index(0)

# Ethernet mode (Network Mode)
# device = ctx.create_net_device("192.168.1.100", 8090)

info = device.get_device_info()
print(f"Device : {info.get_name()}")
print(f"FW     : {info.get_firmware_version()}")
print(f"Serial : {info.get_serial_number()}")

# ── 2. Configure pipeline ──────────────────────────────────────────────
pipeline = Pipeline(device)
config   = Config()

depth_profile = (pipeline
    .get_stream_profile_list(OBSensorType.DEPTH_SENSOR)
    .get_video_stream_profile(640, 576, OBFormat.Y16, 30))
config.enable_stream(depth_profile)

color_profile = (pipeline
    .get_stream_profile_list(OBSensorType.COLOR_SENSOR)
    .get_video_stream_profile(1920, 1080, OBFormat.RGB, 30))
config.enable_stream(color_profile)

config.set_align_mode(obs.OBAlignMode.SW_MODE)   # D2C alignment
pipeline.start(config)

# ── 3. Wait for frameset ───────────────────────────────────────────────
frameset    = pipeline.wait_for_frames(timeout_ms=100)
depth_frame = frameset.get_depth_frame()
color_frame = frameset.get_color_frame()

# ── 4. Read depth buffer ───────────────────────────────────────────────
depth_data  = np.frombuffer(depth_frame.get_data(), dtype=np.uint16)
depth_mat   = depth_data.reshape((depth_frame.get_height(), depth_frame.get_width()))
depth_scale = depth_frame.get_depth_scale()          # mm / unit

print(f"Depth  : {depth_mat.shape}, scale={depth_scale} mm")

# ── 5. Generate point cloud ────────────────────────────────────────────
pc_filter = obs.PointCloudFilter()
pc_filter.set_create_point_format(obs.OBFormat.POINT)
pc_filter.set_calibration_param(pipeline.get_calibration_param())
cloud     = pc_filter.process(frameset)
pts       = np.frombuffer(cloud.get_data(), dtype=np.float32).reshape(-1, 3)
print(f"Points : {len(pts):,}")

# ── 6. Read IMU ────────────────────────────────────────────────────────
accel = device.get_sensor(OBSensorType.ACCEL_SENSOR).get_frame()
gyro  = device.get_sensor(OBSensorType.GYRO_SENSOR).get_frame()

# ── 7. Cleanup ─────────────────────────────────────────────────────────
pipeline.stop()
