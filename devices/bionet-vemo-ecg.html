# ─── Bionet VEMO · BT-Link Next Protocol · Python Example ──────────────────
# Protocol : TCP/IP or USB Serial (BT-Link Next software bridge)
# Docs     : https://bionetus.com/vet/veterinary-multi-parameter-monitors/bt-link-next/
# Data fmt : ISO 11073 / HL7 aECG (XML)

import socket, struct, time

VEMO_HOST = "192.168.1.50"
VEMO_PORT = 3000

# ── 1. Open TCP connection ─────────────────────────────────────────────
sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
sock.connect((VEMO_HOST, VEMO_PORT))
sock.settimeout(5.0)
print("Connected to VEMO BT-Link bridge")

# ── 2. Send handshake / device query ──────────────────────────────────
QUERY_CMD = b'\x02QUERY_DEVICE\x03'     # STX + cmd + ETX
sock.sendall(QUERY_CMD)
resp = sock.recv(1024)
print(f"Device info: {resp.decode('ascii', errors='ignore')}")

# ── 3. Request ECG stream ─────────────────────────────────────────────
START_CMD = b'\x02START_ECG,LEADS=6,SR=500,GAIN=10\x03'
sock.sendall(START_CMD)

# ── 4. Receive ECG data packets ───────────────────────────────────────
# Each packet: [STX][LEN:2B][TIMESTAMP:4B][LEAD_ID:1B][SAMPLES:N×2B][CRC:2B][ETX]
def parse_ecg_packet(data):
    stx = data[0]
    length = struct.unpack('>H', data[1:3])[0]
    timestamp_ms = struct.unpack('>I', data[3:7])[0]
    lead_id = data[7]
    n_samples = (length - 4) // 2
    samples = struct.unpack(f'>{n_samples}h', data[8:8+n_samples*2])
    return {
        "timestamp_ms": timestamp_ms,
        "lead": ["I","II","III","aVR","aVL","aVF"][lead_id],
        "samples_mv": [s / 1000 for s in samples],
        "sample_rate_hz": 500
    }

for _ in range(100):
    raw = sock.recv(4096)
    pkt = parse_ecg_packet(raw)
    print(f"Lead {pkt['lead']} @ {pkt['timestamp_ms']}ms: {len(pkt['samples_mv'])} samples")

# ── 5. Stop & export HL7 aECG record ─────────────────────────────────
sock.sendall(b'\x02STOP_ECG\x03')
sock.sendall(b'\x02EXPORT_HL7,DURATION=10\x03')
hl7_data = b''
while True:
    chunk = sock.recv(8192)
    if not chunk: break
    hl7_data += chunk
with open("vemo_ecg.xml", "wb") as f: f.write(hl7_data)
sock.close()
