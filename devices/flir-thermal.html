# ─── FLIR A40/A50/A70 REST API · Python Example ────────────────────────────
# Base URL: http://{camera_ip}/api
# No auth required on LAN (configure camera firewall if needed)

import requests, json

CAMERA_IP  = "192.168.0.100"
BASE_URL   = f"http://{CAMERA_IP}/api"
TEMP_UNIT  = "C"   # C | F | K

# ── 1. GET current thermal image ───────────────────────────────────────
r = requests.get(f"{BASE_URL}/image/current",
    params={"imgformat": "RJPEG", "width": 640, "height": 480})
print(f"Image {r.status_code}: {len(r.content)} bytes")
with open("thermal.rjpeg", "wb") as f: f.write(r.content)

# ── 2. GET spot temperature measurement ────────────────────────────────
r = requests.get(f"{BASE_URL}/spot/1.json",
    params={"tempUnit": TEMP_UNIT, "pretty": True})
spot = r.json()
print(f"Spot1: {spot['temperature']} {spot['unit']} at ({spot['x']},{spot['y']})")

# ── 3. GET box ROI statistics ───────────────────────────────────────────
r   = requests.get(f"{BASE_URL}/box/1.json", params={"tempUnit": TEMP_UNIT})
box = r.json()
print(f"Box1 : max={box['max']} min={box['min']} avg={box['avg']} {box['unit']}")

# ── 4. GET line temperature profile ────────────────────────────────────
r    = requests.get(f"{BASE_URL}/line/1.json", params={"tempUnit": TEMP_UNIT})
line = r.json()
print(f"Line1: {len(line['points'])} points, max={line['max']}")

# ── 5. GET all alarms ──────────────────────────────────────────────────
r      = requests.get(f"{BASE_URL}/alarms", params={"pretty": True})
alarms = r.json()["alarms"]
for a in alarms:
    print(f"Alarm '{a['name']}': {a['state']} triggered={a['triggered']}")

# ── 6. GET internal temperature sensor ─────────────────────────────────
r      = requests.get(f"{BASE_URL}/tempsensor/1.json", params={"tempUnit": TEMP_UNIT})
sensor = r.json()
print(f"Sensor: {sensor['name']} = {sensor['temperature']} {sensor['unit']}")
