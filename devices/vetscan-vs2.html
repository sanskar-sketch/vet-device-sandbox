# ─── Vetscan VS2 / OptiCell · REST-like API Example ─────────────────────────
# Note: Real devices use Serial/ASTM protocol. This models a REST wrapper.
# SDK    : Custom integration layer (hospital LIS or middleware)
# Protocol: Serial (ASTM E1394) or Ethernet bridge

import requests, json

ANALYZER_URL = "http://192.168.1.50:8080/api/vetscan"

# ── 1. Get analyzer status ────────────────────────────────────────────
r = requests.get(f"{ANALYZER_URL}/status")
status = r.json()
print(f"Model: {status['model']}  Status: {status['state']}")

# ── 2. Submit sample for analysis ─────────────────────────────────────
payload = {
    "patient_id": "PAT-2024-001",
    "species": "canine",
    "rotor_type": "comprehensive_diagnostic",
    "sample_type": "whole_blood",
    "volume_ul": 2.0
}
r = requests.post(f"{ANALYZER_URL}/run", json=payload)
run = r.json()
print(f"Run ID: {run['run_id']}  ETA: {run['estimated_seconds']}s")

# ── 3. Poll for completion ────────────────────────────────────────────
import time
run_id = run["run_id"]
while True:
    r = requests.get(f"{ANALYZER_URL}/run/{run_id}")
    run_status = r.json()
    print(f"Status: {run_status['status']}  Progress: {run_status['progress']}%")
    if run_status["status"] == "completed": break
    time.sleep(5)

# ── 4. Retrieve results ───────────────────────────────────────────────
r = requests.get(f"{ANALYZER_URL}/results/{run_id}")
results = r.json()

print(f"\nResults for {results['patient_id']}:")
for analyte in results["analytes"]:
    flag = analyte["flag"]
    print(f"  {analyte['name']:20} {analyte['value']:>6.2f} {analyte['unit']:8} [{flag}]")

# ── 5. Export to ASTM format ──────────────────────────────────────────
r = requests.get(f"{ANALYZER_URL}/results/{run_id}/astm")
astm_msg = r.text
with open("results.astm", "w") as f: f.write(astm_msg)
